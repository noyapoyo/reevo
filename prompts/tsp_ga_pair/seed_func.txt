def ga_operators_v1():
    def crossover(parent1: list[int], parent2: list[int], rng, ctx: dict) -> list[int]:
        n = len(parent1)
        if n <= 2:
            return parent1[:]
        a, b = sorted(rng.choice(n, size=2, replace=False).astype(int).tolist())
        child = [-1] * n
        child[a:b] = parent1[a:b]
        used = set(child[a:b])
        fill = [x for x in parent2 if x not in used]
        j = 0
        for i in range(n):
            if child[i] == -1:
                child[i] = fill[j]
                j += 1
        return child

    def mutate(route: list[int], rng, ctx: dict) -> list[int]:
        n = len(route)
        if n <= 2:
            return route[:]
        child = route[:]
        i, j = sorted(rng.choice(n, size=2, replace=False).astype(int).tolist())
        child[i], child[j] = child[j], child[i]
        return child

    return {"crossover": crossover, "mutate": mutate}
